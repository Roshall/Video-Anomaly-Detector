#### 安装ffmpeg(编译安装)

sudo apt install yasm libx264-dev libtheora-dev libvorbis-dev libxvidcore-dev  libxext-dev libxfixes-dev libavdevice-dev
apt install build-essential pkg-config

1，解压
unzip FFmpeg-release-6.0.zip

2，切换到根目录
cd FFmpeg-release-6.0


3，执行如下命令（构建编译）
./configure  --prefix=$LIBPATH --enable-pic --enable-shared --enable-asm --enable-gpl --enable-libx264 --enable-nonfree  --enable-pthreads

4，执行编译
make -j$(nproc)

5，安装
make install


~~~

## 安装opencv(编译安装)

~~~
sudo apt install build-essential libgtk2.0-dev
apt install libopenblas-dev

第二步：编译安装

切换到根目录
cd opencv-4.12/opencv

创建构建文件
```bash
mkdir build && cd build

cmake -D CMAKE_BUILD_TYPE=Release \
-D CMAKE_PREFIX_PATH=$LIBPATH \
-D OPENCV_FORCE_3RDPARTY_BUILD=ON \
-D CMAKE_INSTALL_PREFIX=$LIBPATH \
-D OPENCV_EXTRA_MODULES_PATH=../../opencv_contrib-4.12.0/modules \
-D OPENCV_GENERATE_PKGCONFIG=ON \
-D WITH_TBB=OFF \
-D WITH_IPP=OFF \
-D ENABLE_FAST_MATH=1 \
-D OPENCV_ENABLE_NONFREE=ON \
-D INSTALL_C_EXAMPLES=OFF \
-D INSTALL_PYTHON_EXAMPLES=OFF \
-D BUILD_opencv_python3=OFF \
-D BUILD_opencv_python3_tests=OFF \
-D BUILD_TESTS=OFF \
-D BUILD_PERF_TESTS=OFF \
-D BUILD_EXAMPLES=OFF \
-D WITH_CUDA=OFF \
-D WITH-DENABLE_NEON=ON \
-D WITH_FFMPEG=ON \
-D WITH_TIFF=OFF \
-D OPENCV_GENERATE_SETUPVARS=OFF \
-D CMAKE_TLS_VERIFY=0
..
```

make -j$(nproc)

sudo make install 


~~~

#### 安装libevent（编译安装）

~~~

第一步：ubuntu命令行的方式安装libevent依赖库
sudo apt install libssl-dev

第二步：开始编译安装

解压
tar -xzvf  libevent-2.1.12-stable.tar.gz

切换到根目录
cd libevent-2.1.12-stable


mkdir build && cd build

cmake -D CMAKE_BUILD_TYPE=Release -D CMAKE_INSTALL_PREFIX=$LIBPATH -DCMAKE_POLICY_VERSION_MINIMUM=3.5 ..
make -j$(nproc)
sudo make install

~~~
